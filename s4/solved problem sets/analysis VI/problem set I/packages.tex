% Math
\usepackage{mathtools}

\newtheorem{definition}{Definition}[section]
\theoremstyle{plain}
\newtheorem{theorem}{Theorem}[section]
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{corollary}[theorem]{Corollary}

% Computer science: code listings. Use `minted` if enabled in settings.
\ifpsetuseminted
  \usepackage{minted}
\else
  \usepackage{listings}
  \lstset{basicstyle=\ttfamily\small,breaklines=true,breakatwhitespace=true}
\fi
\ifpsetuseminted
  \newenvironment{code}[1][text]{\VerbatimEnvironment\begin{minted}{#1}}{\end{minted}}
\else
  \lstnewenvironment{code}[1][]{}{ }
\fi

% Physics and units
\usepackage{siunitx}
\sisetup{detect-all}
% Load `physics` after siunitx to avoid siunitx warning about \qty; then
% at begin document copy siunitx's \SI to \qty so both work consistently.
\usepackage{physics}
\AtBeginDocument{\RenewCommandCopy\qty\SI}
