\problem{
Let $F$ be defined as: \[ F(x) = \int_{0}^{1} \frac{dt}{(t^{2}+1)(t^{2}+x^{2})} \]
\begin{enumerate}
    \item Show that $F$ is continuous on $(0, +\infty)$.
    \item Deduce the value of the integral:
    \[ \int_{0}^{1} \frac{dt}{(t^{2}+1)^{2}} \]
\end{enumerate}
}

\begin{solution}
\begin{enumerate}
  \item  We have 
  $$ \begin{align}
    f(x,t) = \frac{1}{(x^2 + t^2)(t^2 + 1)}
  \end{align} $$
  $\forall x \in \mathbb{R}_*^+$ and $t \in [0;1]$ $f(x,t)$ is a continuous function then it is integrable.
  we have $$ |f(x,t)| = \left| \frac{1}{(t^2 + 1)(t^2 + x^2)}\right| \le \frac{1}{(t^2 + 1)(t^2 + a^2)} = g(t)$$
  and $g$ is clearly integrable then $F$ is continuous in $(0,\infty)$.

  \item we have $$\begin{align}
      f(x,t) & = \frac{1}{x^2 - 1} \left(\frac{1}{t^2 +1} - \frac{1}{t^2 + x^2}\right) \\
        F(x) & = \frac{1}{x^2 - 1} \left(\int_0^1 \frac{dt}{t^2 +1} - \int_0^1 \frac{dt}{t^2 + x^2}\right) \\
           & = \frac{1}{x^2 - 1} \left(\frac{\pi}{4} - \frac{1}{x}arctan\frac{1}{x}\right) \\ 
  \end{align}$$
  now for $x=1$ we get $$\int_0^1 \frac{dt}{(t^2 + 1)^2} = \lim_{x  \to 1} F(x)  = \frac{\pi}{8} + \frac{1}{4}$$
\end{enumerate}
\end{solution}

\problem{
Let $x > 0$ and $b \neq 0$. We define:
\[ F(x) = \int_{0}^{+\infty} e^{-tx} \sin(bt) \frac{dt}{t} \]
\begin{enumerate}
    \item Show that $F$ is differentiable on $(0, +\infty)$ and calculate $F'(x)$.
    \item Calculate $\lim_{x \to +\infty} F(x)$ and $\lim_{x \to 0} F(x)$.
    \item Deduce the value of the integral:
    \[ \int_{0}^{+\infty} \frac{\sin(bt)}{t} dt \]
\end{enumerate}
}

\begin{solution}
\begin{enumerate}
  \item we have $$\begin{align}
      f(x,t) & = \frac{e^{-tx}S_{bt}}{t} \\
      \frac{\partial f}{\partial x} & = e^{-tx}S_{bt}
  \end{align}$$
  $\forall x > a > 0$ $$\left|\frac{\partial f}{\partial x}\right| = |e^{-tx}S_{bt}| \le e^{-ta}$$
  $\int_0^{\infty} e^{-at} dt$ converges for $a > 0$.
  $$\begin{align}
    F'(x) & = \int_0^{\infty} - e^{-tx} S_{bt}dt \\
          & = \frac{-b}{x^2 + b^2}
  \end{align}$$
\item \begin{enumerate}
    \item  $x \to \infty$ : $$\lim_{\infty} F(x) = \int_0^{\infty}\lim_{\infty} e^{-tx} \frac{S_{bt}}{t} dt$$
    \item $x \to 0$ : $$\lim_{0}  = F(0) = \int_0^{\infty} \frac{S_{bt}}{t}dt$$ 
  \end{enumerate} 
  
\item $$\begin{align}
    F(x) & = \int \frac{-b}{x^2 + b^2}dx = - arctan \frac{x}{b} + C \\
    0 & = \lim_{\infty} (- arctan \frac{x}{b} + C) \\
     & \begin{cases}
       \text{if } b > 0 \implies C = \frac{\pi}{2}\\
       \text{if } b < 0 \implies C = -\frac{\pi}{2}
    \end{cases} \\ 
    F(0) & = \int_0^{\infty} \frac{S_{bt}}{t} dt = \frac{\pi}{2}(-1)^n
\end{align}$$ 
\end{enumerate}
\end{solution}

\problem{
Consider the functions $f$ and $g$ defined on $\mathbb{R}^{+}$ by:
\[ f(x) = \int_{0}^{\infty} \frac{e^{-xt}}{1+t^{2}} dt \quad \text{and} \quad g(x) = \int_{0}^{\infty} \frac{\sin t}{t+x} dt \]
\begin{enumerate}
    \item Show that $f$ and $g$ are of class $C^{2}$ on $\mathbb{R}^{+*}$ and that they satisfy the differential equation:
    \[ y'' + y = \frac{1}{x} \]
    \item Show that $f$ and $g$ are continuous at 0.
    \item Show that $\lim_{x \to +\infty} f(x) = \lim_{x \to +\infty} g(x) = 0$.
    \item Deduce that $g(x) = f(x)$, then give the value of the integral:
    \[ \int_{0}^{\infty} \frac{\sin t}{t} dt \]
\end{enumerate}
}

\begin{solution}
\begin{enumerate}
  \item 
    \begin{enumerate}
      \item $$\begin{align}
          f'(x) & = \int_0^{\infty} \frac{e^{-xt}}{1+t^2}dt \\
          f"(x) & = \int_0^{\infty} \frac{t^2 e^{-xt}}{1+t^2}dt \\
          f"(x) & + f(x) = \frac{1}{x}
      \end{align}$$ 
      
    \item $$\begin{align}
        g'(x) & = \int_0^{\infty} \frac{S_t}{t+x} dt \\
        g"(x) & = \int_0^{\infty} \frac{2S_t}{(t+x)^3}dt \\
        g"(x) & + g(x) = \frac{1}{x}
    \end{align}$$ 
    \end{enumerate}
  
  \item \begin{enumerate}
      \item $f(0) = \int_0^{\infty} \frac{1}{1+t^2} dt = arctan(t)|_0^{\infty} = \frac{\pi}{2}$; it converges uniformaly to $0$. 
      
      \item $g(0) = \int_0^{\infty} \frac{S_t}{t}dt$ converges conditionally, $g$ is continuous at $0$.
  \end{enumerate} 
  
\item $\lim_{\infty} f(x) = 0$; $|g(x)| \le \int_0^{\infty} \frac{|S_t|}{t+x}dt \implies \lim_{\infty} g(x) = 0$. 
  
\item 
  \begin{enumerate}
    \item both satify the differential equation, their difference $h(x) = f(x) - g(x)$ satisfies $h" - h = 0$ meaning $h(x) = AC_x + BS_x$.

    \item since $\lim_{\infty} f(x) = \lim_{\infty} g(x) = 0 \implies A = B = 0 \implies f(x) = g(x)$
   
    \item since $f(0) = g(0)$ $$\int_0^{\infty} \frac{S_t}{t} dt = \frac{\pi}{2}$$ 
  \end{enumerate}
\end{enumerate}
\end{solution}
