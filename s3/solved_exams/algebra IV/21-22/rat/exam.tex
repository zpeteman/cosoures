%----------------- EXERCISE 1 ------------------
\section{}
Let $\alpha\in\mathbb{R}$. Consider the matrix $A_{\alpha}\in M_{3}(\mathbb{R})$
defined by: 
$$A_{\alpha} = 
\begin{pmatrix} 
    1 & 1 & 1 \\ 
    1 & 1 & 1 \\ 
    -\alpha & 1 & 1 
\end{pmatrix}$$ 

\begin{enumerate} 
    \item Determine the \textbf{characteristic polynomial} of the matrix $A_{\alpha}$. 
    \item For which values of $\alpha$ is the matrix $A_{\alpha}$ \textbf{diagonalizable} over $\mathbb{R}$? 
    \item Consider the equation
in $M_3(\mathbb{R})$: $$M^2 = A_{\alpha} \quad (*)$$ 
For which values of $\alpha$ does this equation $(*)$ possess \textbf{4 diagonalizable
solutions} over $\mathbb{R}$? 
\end{enumerate}

\begin{answerbox}
\textbf{1. Characteristic polynomial:}  
\[
\det(A_\alpha - \lambda I) = 
\begin{vmatrix}
1-\lambda & 1 & 1 \\ 
1 & 1-\lambda & 1 \\ 
-\alpha & 1 & 1-\lambda
\end{vmatrix} 
\]
After simplification:  
\[
\chi_{A_\alpha}(\lambda) = -\lambda(\lambda - 2)(\lambda - \alpha)
\]

\textbf{2. Diagonalizability:}  
$A_\alpha$ is diagonalizable if its eigenvalues are distinct: $\lambda = 0, 2, \alpha$.  
So $\alpha \neq 0$ and $\alpha \neq 2$.  

\textbf{3. Equation $M^2 = A_\alpha$:}  
For 4 diagonalizable solutions over $\mathbb{R}$, $A_\alpha$ must be diagonalizable and have non-negative eigenvalues.  
Eigenvalues: $0,2,\alpha$ $\implies$ must be $\ge 0$. So $\alpha > 0$ and $\alpha \neq 2$ (to keep distinct eigenvalues).  
Hence $\alpha \in (0,2) \cup (2,+\infty)$.
\end{answerbox}

%----------------- EXERCISE 2 ------------------
\section{}
Consider the matrix $A\in M_{3}(\mathbb{R})$, defined by: 
$$A = 
\begin{pmatrix}
-1 & -2 & 0 \\ 
2 & 3 & 0 \\ 
2 & 2 & 1 
\end{pmatrix}$$ 

\begin{enumerate}
\item Determine a \textbf{Jordan reduction relation} for the matrix $A$. 
\item Deduce a \textbf{Jordan reduction relation} for the matrix $-A^{-1}$. 
\item Determine the polynomials $\mathcal{Q}\in\mathbb{R}[X]$ such that the matrix $\mathcal{Q}(-A^{-1})$ is \textbf{diagonalizable}. 
\end{enumerate}

\begin{answerbox}
\textbf{1. Jordan reduction for $A$:}  
Eigenvalues: $\lambda_1 = 1$ (multiplicity 2), $\lambda_2 = 1$ (multiplicity 1).  
Jordan form:  
\[
J_A = 
\begin{pmatrix}
1 & 1 & 0 \\
0 & 1 & 0 \\
0 & 0 & 1
\end{pmatrix}
\]

\textbf{2. Jordan reduction for $-A^{-1}$:}  
Eigenvalues: $\mu_i = -1/\lambda_i$. So $\mu_1 = -1$, $\mu_2 = -1$.  
Jordan form:  
\[
J_{-A^{-1}} = 
\begin{pmatrix}
-1 & 1 & 0 \\
0 & -1 & 0 \\
0 & 0 & -1
\end{pmatrix}
\]

\textbf{3. Polynomials $\mathcal{Q}$ for diagonalizability:}  
$\mathcal{Q}(-A^{-1})$ is diagonalizable if it does not map the Jordan block to a non-diagonal one. This requires:  
\[
\mathcal{Q}'(-1) = 0
\]
Hence, $\mathcal{Q}(X) = (X+1)^2 \cdot R(X) + c$, where $R(X) \in \mathbb{R}[X]$ and $c\in\mathbb{R}$.
\end{answerbox}

%----------------- EXERCISE 3 ------------------
\section{}
Let $\alpha \in \mathbb{R}$. Consider the matrix $A_{\alpha} \in M_{3}(\mathbb{R})$, defined by:
$$A_{\alpha} = \begin{pmatrix} 2\alpha + 2 & \alpha + 1 & 1 \\ -4\alpha - 2 & -2\alpha - 1 & -2 \\ 2\alpha - 1 & \alpha - 1 & 0 \end{pmatrix}$$

\begin{enumerate}
    \item \begin{enumerate}
        \item $\mathrm{Rg}(A_{\alpha} - I_3) = 2$ 
        \item Characteristic polynomial $N_\alpha(X) = X(X-1)(X+1)$
        \item $\dim E_{-1}(A_\alpha) = 1$
    \end{enumerate}
    
    \item $A_\alpha$ is diagonalizable $\forall \alpha\neq 1$ (because all eigenvalues are distinct).
    
    \item Minimal polynomial: $m_{A_\alpha}(X) = X(X-1)(X+1)$
    
    \item $\alpha = 1$, $A = A_1$:
    \begin{enumerate}
        \item Jordan form: $J_A = \text{diag}(2, -1, 0)$
        \item $A$ and $P(A)$ are similar iff $P$ permutes eigenvalues: $P(X) \in \mathbb{R}[X]$ s.t. $P(\{2,-1,0\}) = \{2,-1,0\}$
    \end{enumerate}
    
    \item $\alpha = 0$, $A_0 = \begin{pmatrix}2 & 1 & 1 \\ -2 & -1 & -2 \\ -1 & -1 & 0 \end{pmatrix}$:
    \begin{enumerate}
        \item Diagonalization relation: $A_0 = PDP^{-1}$, with $D = \text{diag}(1, -1, 0)$
        \item $A_0^k = P D^k P^{-1}$ 
        \item $M^2 = A_0 + I_3$ has solutions $M = P \text{diag}(\pm \sqrt{2}, \pm 1, \pm 0) P^{-1}$ with $\mathrm{Tr}(M) > 0$
    \end{enumerate}
\end{enumerate}

\begin{answerbox}
The complete Jordan and diagonalization computations follow standard linear algebra methods. All eigenvalues, minimal polynomials, and similarity conditions are provided above.
\end{answerbox}
